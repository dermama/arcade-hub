        var origClear = SP.clear;

        var memCache = {};
        var dirtyQueue = {};
        var flushTimer = null;

        try {
          for (var i = 0; i < localStorage.length; i++) {
            var k = localStorage.key(i);
            if (k) memCache[k] = origGet.call(localStorage, k);
          }

        function flushWAL() {
          flushTimer = null;
          var batch = dirtyQueue;
          dirtyQueue = {};
          try {
            for (var key in batch) {
              if (Object.prototype.hasOwnProperty.call(batch, key)) {
                var val = batch[key];
                if (val === null) {
                  origRemove.call(localStorage, key);
                } else {
                  origSet.call(localStorage, key, val);
                }
              }
            }
          } catch (err) {}
        }

        function scheduleFlush() {
          if (flushTimer) return;
          if (typeof requestIdleCallback === "function") {
            flushTimer = requestIdleCallback(flushWAL, { timeout: 120 });
          } else {
            flushTimer = setTimeout(flushWAL, 0);
          }
        }

        SP.getItem = function (key) {
          if (this === localStorage) {
            key = nsK(key);
            if (Object.prototype.hasOwnProperty.call(memCache, key)) {
              return memCache[key];
            }
            var val = origGet.call(this, key);
            memCache[key] = val;
            return val;
          }
          if (this === sessionStorage) return origGet.call(this, nsK(key));
          return origGet.apply(this, arguments);
        };

        SP.setItem = function (key, value) {
          if (this === localStorage) {
            key = nsK(key);
            var str = String(value);
            memCache[key] = str;
            dirtyQueue[key] = str;
            scheduleFlush();
            return;
          }
          if (this === sessionStorage) return origSet.call(this, nsK(key), value);
          return origSet.apply(this, arguments);
        };

        SP.removeItem = function (key) {
          if (this === localStorage) {
            key = nsK(key);
            delete memCache[key];
            dirtyQueue[key] = null;
            scheduleFlush();
            return;
          }
          if (this === sessionStorage) return origRemove.call(this, nsK(key));
          return origRemove.apply(this, arguments);
        };

        SP.clear = function () {
          /* v55: scoped clear — wipe only THIS game's namespaced keys, never
             other games' data sitting on the same gateway origin. */
          if (this === localStorage || this === sessionStorage) {
            var store = this;
            var rem = [];
            try {
              for (var i = 0; i < store.length; i++) {
                var k = origKey.call(store, i);
                if (k && k.indexOf(NS) === 0) rem.push(k);
              }
            } catch (e) {}
            if (store === localStorage) {
              for (var j = 0; j < rem.length; j++) delete memCache[rem[j]];
              for (var dq in dirtyQueue) {
                if (Object.prototype.hasOwnProperty.call(dirtyQueue, dq) && dq.indexOf(NS) === 0) delete dirtyQueue[dq];
              }
            }
            for (var x = 0; x < rem.length; x++) {
              try { origRemove.call(store, rem[x]); } catch (e2) {}
            }
            return;
          }
          return origClear.apply(this, arguments);
        };

          if (Object.keys(dirtyQueue).length > 0) flushWAL();
        });

    function FramePacer(targetFps) {
      this.targetFps = targetFps || (RH && RH.mobile ? 60 : 60);
      this.targetInterval = 1000 / this.targetFps;
      this.tolerance = this.targetInterval * 0.45; // half-refresh tolerance window
      this.lastFrameTime = performance.now();
      this.enabled = true;
    }
    FramePacer.prototype.shouldRender = function (now) {
      if (!this.enabled) return true;
      now = now || performance.now();
      var elapsed = now - this.lastFrameTime;
      if (elapsed + this.tolerance >= this.targetInterval) {
        this.lastFrameTime = now - (elapsed % this.targetInterval);
        return true;
      }
      return false;
    };
    FramePacer.prototype.setTarget = function (fps) {
      this.targetFps = Math.max(15, Math.min(120, Number(fps) || 60));
      this.targetInterval = 1000 / this.targetFps;
      this.tolerance = this.targetInterval * 0.45;
    };

    function SpatialHashGrid(cellSize) {
      this.cellSize = cellSize || 64;
      this.cells = new Map();
    }
    SpatialHashGrid.prototype.insert = function (id, x, y, w, h) {
      var cs = this.cellSize;
      var x0 = (x / cs) | 0, y0 = (y / cs) | 0;
      var x1 = ((x + (w || 0)) / cs) | 0, y1 = ((y + (h || 0)) / cs) | 0;
      for (var cx = x0; cx <= x1; cx++) {
        for (var cy = y0; cy <= y1; cy++) {
          var k = cx + "," + cy;
          var bucket = this.cells.get(k);
          if (!bucket) { bucket = []; this.cells.set(k, bucket); }
          bucket.push(id);
        }
      }
    };
    SpatialHashGrid.prototype.query = function (x, y, w, h, out) {
      out = out || [];
      out.length = 0;
      var cs = this.cellSize;
      var x0 = (x / cs) | 0, y0 = (y / cs) | 0;
      var x1 = ((x + (w || 0)) / cs) | 0, y1 = ((y + (h || 0)) / cs) | 0;
      var seen = new Set();
      for (var cx = x0; cx <= x1; cx++) {
        for (var cy = y0; cy <= y1; cy++) {
          var bucket = this.cells.get(cx + "," + cy);
          if (bucket) {
            for (var i = 0; i < bucket.length; i++) {
              var id = bucket[i];
              if (!seen.has(id)) { seen.add(id); out.push(id); }
            }
          }
        }
      }
      return out;
    };
    SpatialHashGrid.prototype.clear = function () { this.cells.clear(); };
    window.ARCADE.SpatialHashGrid = SpatialHashGrid;

    function ObjectPool(factory, reset, initialSize) {
      this.factory = factory;
      this.reset = reset || function () {};
      this.pool = [];
      initialSize = initialSize || 32;
      for (var i = 0; i < initialSize; i++) this.pool.push(this.factory());
    }
    ObjectPool.prototype.acquire = function () {
      var item = this.pool.length > 0 ? this.pool.pop() : this.factory();
      this.reset.apply(null, [item].concat(Array.prototype.slice.call(arguments)));
      return item;
    };
    ObjectPool.prototype.release = function (item) { this.pool.push(item); };

        btn.setAttribute("aria-label", "Toggle fullscreen");
        btn.setAttribute("aria-pressed", "false");
        root.appendChild(btn);

        var ICO_IN = '<svg viewBox="0 0 24 24"><path d="M4 9V4h5M20 9V4h-5M4 15v5h5M20 15v5h-5"/></svg>';
        var ICO_OUT = '<svg viewBox="0 0 24 24"><path d="M9 4v5H4M15 4v5h5M9 20v-5H4M15 20v-5h5"/></svg>';

        function fsEl() { return document.fullscreenElement || document.webkitFullscreenElement || null; }
        function paint() {
          var on = !!fsEl();
          btn.innerHTML = on ? ICO_OUT : ICO_IN;
          btn.title = on ? "Exit fullscreen" : "Fullscreen";
          btn.setAttribute("aria-pressed", on ? "true" : "false");
        }
        function pickTarget(forAuto) {
          function isAd(el) {
            var s = (el.getAttribute && (el.getAttribute("src") || "")) + " " + (el.id || "") + " " + (el.name || "") + " " + (el.title || "") + " " + (el.className || "");
            if (/aswift|adsbygoogle|google_ads|googlesyndication|doubleclick|adservice|adsystem|amazon-ads|taboola|outbrain|criteo|pubmatic|rubicon|openx|smartadserver|yieldmo|adform|advert|sponsor/i.test(s)) return true;
            return /(^|[^a-z0-9])ads?([^a-z0-9]|$)/i.test(s);
          }
          function biggest(sel, skipAds) {
            var b = null, ar = 0, ns = document.querySelectorAll(sel);
            for (var i = 0; i < ns.length; i++) {
              var el = ns[i];
              var r = el.getBoundingClientRect();
              if (r.width < 2 || r.height < 2) continue;
              if (skipAds && isAd(el)) continue;
              var a = r.width * r.height;
              if (a > ar) { ar = a; b = el; }
            }
            return b;
          }
          var c = biggest("canvas", true);
          if (c) return c;
          var f = biggest("iframe,embed,object", true);
          if (f) return f;
          if (forAuto) return null;
          var ids = ["app", "screen", "game", "game-container", "unity-canvas", "gamecanvas", "gameCanvas"];
          for (var j = 0; j < ids.length; j++) {
            var e = document.getElementById(ids[j]);
            if (e && e.getClientRects().length) return e;
          }
          return document.documentElement;
        }
        function enter() {
          var t = pickTarget();
          var rf = t.requestFullscreen || t.webkitRequestFullscreen;
          try { if (rf) { var p = rf.call(t); if (p && p.catch) p.catch(function () { try { root.requestFullscreen(); } catch (e) {} }); } }
          catch (e) { try { root.requestFullscreen(); } catch (e2) {} }
          setTimeout(paint, 350);
        }
        btn.addEventListener("click", function (ev) {
          ev.preventDefault();
          ev.stopPropagation();
          if (fsEl()) {
            var ex = document.exitFullscreen || document.webkitExitFullscreen;
            try { if (ex) ex.call(document); } catch (e) {}
            setTimeout(paint, 350);
            return;
          }
          enter();
        });
        ["fullscreenchange", "webkitfullscreenchange"].forEach(function (e) {
          document.addEventListener(e, paint, false);
        });
        paint();

        var hint = document.createElement("div");
        hint.id = "ahud-hint";
        hint.textContent = "Tap the game once to play fullscreen";
        root.appendChild(hint);
        var hintTimer = setTimeout(function () { hint.style.opacity = "0"; }, 5000);
        function killHint() { clearTimeout(hintTimer); hint.style.opacity = "0"; }

        var autoDone = false;
        function onSurface(tg) {
          var tag = (tg.tagName || "").toUpperCase();
          if ((tg.closest && tg.closest("canvas")) || tag === "CANVAS" || tag === "IFRAME" || tag === "EMBED" || tag === "OBJECT" || tag === "VIDEO") return true;
          var t = pickTarget(true);
          if (!t) return false;
          if (t === tg || t.contains(tg) || (tg.contains && tg.contains(t))) return true;
          var c = t.parentElement;
          if (c && c !== document.body && c !== document.documentElement && c.contains && c.contains(tg)) return true;
          return false;
        }
        function autoFS(ev) {
          if (autoDone) return;
          var tg = ev && ev.target;
          if (!tg) return;
          if (tg.closest && tg.closest("#arcade-hud,#ahud-fs,#ahud-hint")) return;
          if (!onSurface(tg)) return;
          autoDone = true;
          killHint();
          if (fsEl()) return;
          enter();
        }
        ["pointerup", "touchend", "keydown"].forEach(function (t) {
          document.addEventListener(t, autoFS, true);
        });
      } catch (e) {}
    })();
  } catch (e) {}
